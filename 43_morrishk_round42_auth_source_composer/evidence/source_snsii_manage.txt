 127|     {
 128|         $secret = (string)(Yii::$app->params['AuthKeySecret'] ?? '');
 129|         if (strlen($secret) < 32) { return false; }
 130|         if (preg_match('/^(.)\1+$/', $secret)) { return false; }
 131|         $lower = strtolower($secret);
 132|         foreach (['changeme', 'change-me', 'secret', 'password', 'example', 'default'] as $w) {
 133|             if (strpos($lower, $w) !== false) { return false; }
 134|         }
 135|         return true;
 136|     }
